\begin{abstract}
We resolve both main conjectures of Z\'u\~niga-Galindo (\href{https://arxiv.org/abs/2601.19070}{arXiv:2601.19070}, January~2026).
First, we prove and sharpen Conjecture~\texttt{Conj\_tree}: for $\phi = \tanh$, zero input, and $T_W = \sum_{j \ge 0} c_j P_j$ ($c_j \ge 0$) with $A_l = \sum_{j \ge l} c_j > 1$ and $\beta_l = \sum_{j < l} c_j < M_{A_l} = A_l \tanh s_{A_l} - s_{A_l}$, the level-$l$ space $D^l(\mathbb{Z}_p)$ contains \emph{exactly} $3^{p^l}$ equilibria---one per three-branch pattern---each hyperbolic in $D^l(\mathbb{Z}_p)$ with instability index equal to its number of middle-branch coordinates, and all $2^{p^l}$ outer-only equilibria are linearly stable in the full space $L^2(\mathbb{Z}_p)$.
The core step is a weighted Gershgorin estimate establishing positive definiteness of the outer Hessian block all the way up to the fold bound $\beta_l < M_{A_l}$.
Second, we refute Conjecture~\texttt{Conj\_random} as stated for general fixed nonnegative profiles: for $p = 2$, we construct a single deterministic, finite-valued, nonnegative profile $v$ with $v(|\cdot|_2) \in L^1(\mathbb{Z}_2)$ and $0 < \rho(v)^2 \le 1/4096$ ($\rho(v) \le 1/64 < 1$) along which an infinite sequence of dyadic widths $N_k = 2^{l_k}$ has at least three equilibria with probability at least $\frac{1}{2} - 2^{-k}$.
The profile $v$ is unbounded as $|x|_2 \to 0$; bounded fixed profiles ($v \in L^\infty$) and the supercritical ($\rho(v) > 1$) implication remain open.
\end{abstract}

\section{Proof and Sharpening of Conjecture~\texttt{Conj\_tree}}

Normalize Haar measure on $\mathbb{Z}_p$ to mass~$1$. Let $G_l := \mathbb{Z}_p / p^l \mathbb{Z}_p$ index the $n := p^l$ balls of measure $p^{-l}$, let $D^l := D^l(\mathbb{Z}_p) \subset L^2(\mathbb{Z}_p)$ be the $n$-dimensional space of functions constant on these balls, and let $P_j f(x) := p^j \int_{x + p^j \mathbb{Z}_p} f(y)\,dy$ be the orthogonal projection onto~$D^j$.
For a nonnegative nonincreasing radial kernel with $T_W = \sum_{j \ge 0} c_j P_j$ ($c_j \ge 0$, $\sum_{j \ge 0} c_j < \infty$), define
\[
A := A_l = \sum_{j \ge l} c_j, \qquad \beta := \beta_l = \sum_{j < l} c_j.
\]
The zero-input dynamics is $\dot{h} = -h + T_W \tanh h$. At an equilibrium~$h$, linear stability in $L^2(\mathbb{Z}_p)$ means that the spectrum of $-I + T_W D$ (where $D$ is multiplication by $\sech^2 h$) lies strictly in $\{\operatorname{Re} z < 0\}$.

On $D^l$, we have $P_j|_{D^l} = I$ for all $j \ge l$, so $T_W|_{D^l} = AI + B$, where $B := \sum_{j < l} c_j P_j|_{D^l}$ is symmetric, positive semidefinite, entrywise nonnegative, and has constant row sums~$\beta$.
In rate coordinates $u_i := \tanh h_i \in (-1, 1)$, the equilibrium equation on $D^l$ reads
\[
f(u_i) = (Bu)_i \quad (i = 1, \dots, n), \qquad f(t) := \atanh t - At \quad (-1 < t < 1).
\]
For $A > 1$, set $s_A := \arcosh\sqrt{A} > 0$ and $r := \tanh s_A = \sqrt{1 - 1/A} \in (0, 1)$, so $A = (1 - r^2)^{-1}$.
The odd function $f$ is strictly convex on $(0, 1)$, strictly decreasing on $[-r, r]$, and strictly increasing on $[r, 1)$, with fold height
\[
M := M_A = A\tanh s_A - s_A = Ar - s_A = -f(r) > 0.
\]
Let $I_- := (-1, -r)$, $I_0 := (-r, r)$, and $I_+ := (r, 1)$ be the three open branch intervals.

\begin{theorem}[Exact tree-indexed equilibria and $L^2(\mathbb{Z}_p)$ linear stability]\label{thm:tree}
Assume $A > 1$ and $\beta < M_A$. Then:
\begin{enumerate}[label=\textup{(\arabic*)},nosep]
\item For each branch pattern $\sigma \in \{-, 0, +\}^n$, there exists a unique equilibrium $u \in D^l$ with $u_i \in I_{\sigma_i}$ for all~$i$. In particular, $D^l$ contains \emph{exactly} $3^{p^l}$ equilibria.
\item Every equilibrium $u \in D^l$ is hyperbolic in $D^l$ with instability index $|\{i : \sigma_i = 0\}|$, giving exactly $\binom{p^l}{k} 2^{p^l - k}$ equilibria of index~$k$.
\item All $2^{p^l}$ outer-only equilibria ($\sigma \in \{-, +\}^n$) are linearly stable in the full space $L^2(\mathbb{Z}_p)$, whereas every remaining equilibrium in $D^l$ has a positive eigenvalue already in $D^l$.
\end{enumerate}
\end{theorem}

\begin{proof}
\emph{Step~1: Existence on each branch pattern.}
Since $\beta < M$, each branch of $f$ has a continuous inverse $g_-, g_0, g_+ : [-\beta, \beta] \to (-1, 1)$ taking values in $I_-, I_0, I_+$, respectively. For any pattern $\sigma \in \{-, 0, +\}^n$, since $\|Bu\|_\infty \le \beta$ for all $u \in [-1, 1]^n$, the continuous map $u \mapsto (g_{\sigma_i}((Bu)_i))_{i=1}^n$ sends $[-1, 1]^n$ into itself. Brouwer's fixed-point theorem supplies at least one equilibrium per pattern (including when $\beta = 0$).

\emph{Step~2: Weighted Gershgorin estimate on the outer block.}
Let $u \in D^l$ be any equilibrium. Since $\|Bu\|_\infty \le \beta < M$, no coordinate satisfies $|u_i| = r$, so $O := \{i : |u_i| > r\}$ and $C := \{i : |u_i| < r\}$ partition $\{1, \dots, n\}$. Define the symmetric matrix
\[
H := \diag(f'(u_i)) - B.
\]
We prove that the principal submatrix $H_{OO}$ is strictly positive definite. For $i \in O$, put $t_i := |u_i| \in (r, 1)$, $w_i := \min(t_i - r, 1 - t_i) > 0$, $m := (1 + r)/2$, and $d := (1 - r)/2$, so $0 < w_i \le d$. We claim that
\begin{equation}\label{eq:gershgorin}
f'(t_i)\,w_i > \sum_{j \in O} B_{ij} w_j \qquad (\forall i \in O).
\end{equation}
If $t_i \le m$, then $w_i = t_i - r$. By strict convexity of $f$ on $(r, 1)$ and oddness of~$f$,
\[
f'(t_i)(t_i - r) > f(t_i) - f(r) = M + \sgn(u_i)(Bu)_i > \beta + \sgn(u_i)(Bu)_i = \sum_{j=1}^n B_{ij}\bigl(1 + \sgn(u_i)u_j\bigr).
\]
Because $B_{ij} \ge 0$ and $|u_j| < 1$, every term is nonnegative; restricting the sum to $j \in O$ and noting $1 + \sgn(u_i)u_j \ge 1 - |u_j| \ge w_j$ establishes~\eqref{eq:gershgorin}.

If $t_i \ge m$, then $w_i = 1 - t_i$. On $[r, 1)$, the function $(1 - t)f'(t) = (1 + t)^{-1} - A(1 - t)$ has derivative $A - (1 + t)^{-2} > 1 - 1 = 0$, so it is strictly increasing. Substituting $A = (1 - r^2)^{-1}$ gives
\[
f'(m) = \frac{4}{(1 - r)(3 + r)} - \frac{1}{1 - r^2} = \frac{1 + 3r}{(1 - r^2)(3 + r)} = Ar + \frac{1}{3 + r} > Ar - s_A = M > \beta.
\]
Hence $f'(t_i)(1 - t_i) \ge (1 - m)f'(m) = d f'(m) > d\beta \ge \sum_{j \in O} B_{ij} w_j$, proving~\eqref{eq:gershgorin} for all $i \in O$.

Setting $W := \diag(w_i)_{i \in O}$, every Gershgorin disk of the similar matrix $W^{-1} H_{OO} W$ lies strictly in $(0, \infty)$. Since $H_{OO}$ is real symmetric, it follows that $H_{OO} \succ 0$.

\emph{Step~3: Hyperbolicity and instability index.}
For $i \in C$ we have $f'(u_i) < 0$, and $B_{CC} \succeq 0$, so $H_{CC} \prec 0$. Eliminating $H_{OO} \succ 0$ yields the negative definite Schur complement $H_{CC} - H_{CO} H_{OO}^{-1} H_{OC} \prec 0$ (the one-block cases $C = \emptyset$ and $O = \emptyset$ are direct). By Sylvester's law of inertia, $H$ is invertible with exactly $|C|$ negative and $|O|$ positive eigenvalues. In leaf coordinates, $D = \diag(1 - u_i^2)$ satisfies $f'(u_i) = D_{ii}^{-1} - A$, so $H = D^{-1} - (AI + B)$. Thus the Jacobian $J = -I + (AI + B)D$ satisfies
\[
D^{1/2} J D^{-1/2} = -I + D^{1/2}(AI + B)D^{1/2} = -D^{1/2} H D^{1/2},
\]
so $J$ is hyperbolic with exactly $|C|$ positive and $|O|$ negative eigenvalues.

\emph{Step~4: Homotopy uniqueness.}
Replace $B$ by $bB$ for $b \in [0, 1]$. Every solution $h = (AI + bB)\tanh h$ satisfies $\|h\|_\infty \le A + \beta$, confining $u$ to the compact cube $[-\tanh(A + \beta), \tanh(A + \beta)]^n \subset (-1, 1)^n$, while $|(bBu)_i| \le \beta < M$ keeps every coordinate strictly away from $\pm r$. By Step~3, the derivative $H$ of $u \mapsto f(u) - bBu$ is invertible at every solution for every $b \in [0, 1]$. Consequently, each fiber over~$b$ is finite, each root continues uniquely locally by the implicit function theorem, and compactness prevents additional roots from accumulating onto the fiber. The root count in each branch pattern is therefore locally constant, hence constant on $[0, 1]$. At $b = 0$, $f(u_i) = 0$ has exactly one root per branch, yielding exact cardinality $3^n$ and $\binom{n}{k} 2^{n - k}$ states of index~$k$.

\emph{Step~5: Full $L^2(\mathbb{Z}_p)$ linear stability.}
At any equilibrium $h \in D^l$, the multiplication operator $D = \sech^2 h$ is constant on each level-$l$ ball and thus commutes with $P_l$; every $P_j$ commutes with $P_l$ as well. Hence $D^l$ and $(D^l)^\perp$ are invariant subspaces of $-I + T_W D$. On $(D^l)^\perp$, every $P_j$ with $j < l$ vanishes, so $\|T_W|_{(D^l)^\perp}\|_{\mathrm{op}} \le \sum_{j \ge l} c_j = A$. For an outer-only equilibrium ($C = \emptyset$), every $|u_i| > r = \sqrt{1 - 1/A}$, whence $\|D\|_{\mathrm{op}} = \max_i(1 - u_i^2) < 1/A$. Therefore $\|T_W D|_{(D^l)^\perp}\|_{\mathrm{op}} \le A\|D\|_{\mathrm{op}} < 1$, so the complementary spectrum lies inside the disk $\{|z + 1| \le A\|D\|_{\mathrm{op}}\} \subset \{\operatorname{Re} z < 0\}$. Together with Step~3 on $D^l$, all $2^n$ outer-only equilibria are linearly stable in $L^2(\mathbb{Z}_p)$.
\end{proof}

\begin{remark}
Theorem~\ref{thm:tree} settles Conjecture~\texttt{Conj\_tree} affirmatively and determines the exact equilibrium count inside $D^l(\mathbb{Z}_p)$; it makes no assertion about additional equilibria lying outside $D^l(\mathbb{Z}_p)$.
\end{remark}

\section{Fixed-Profile Counterexample to Conjecture~\texttt{Conj\_random}}

At level~$l$, let $M^{(l)} \in \mathbb{R}^{p^l \times p^l}$ have independent entries $M^{(l)}_{IK} \sim \mathcal{N}(0, p^{-l} v(|I - K|_p))$ for a fixed nonnegative profile~$v$, with equilibria $h = M^{(l)}\tanh h$ and $\rho(v)^2 := \lim_{l \to \infty} \sum_{K \in G_l} p^{-l} v(|K|_p)$. Conjecture~\texttt{Conj\_random} asserts that with probability tending to~$1$ as $l \to \infty$, $\rho(v) < 1$ implies $h = 0$ is the unique equilibrium, while $\rho(v) > 1$ implies exponentially many equilibria. Throughout this section, $\|\cdot\|$ denotes Euclidean operator norm and $\log$ is natural logarithm.

\begin{lemma}[Deterministic Brouwer degree test]\label{lem:degree}
For any real $n \times n$ matrix $M$, if $\det(I - M) < 0$, then $h = M\tanh h$ has at least three solutions in $\mathbb{R}^n$.
\end{lemma}

\begin{proof}
Let $F(h) := h - M\tanh h$ and $R := 1 + \|M\|\sqrt{n}$. On $\|h\|_2 = R$, $\|b M\tanh h\|_2 \le \|M\|\sqrt{n} < R$ for all $b \in [0, 1]$, so the homotopy $h - b M\tanh h$ is nonvanishing on $\partial B_R(0)$, giving $\deg(F, B_R(0), 0) = 1$ with all zeros interior. Since $DF(0) = I - M$ has negative determinant, the local index at~$0$ is $-1 \neq 1$. Additivity of Brouwer degree forces a nonzero root $h^*$, and oddness yields the distinct root $-h^*$.
\end{proof}

Take $p = 2$. An active shell at distance $2/N_j = 2^{-(l_j - 1)}$ (where $N_j = 2^{l_j}$) receives value $v(2/N_j) := N_j a_j$ (and $v(0) := 0$). At dyadic width $n = 2^l \ge N_j$, each row $I \in G_l$ has exactly $n/N_j$ entries $K \in G_l$ with $|I - K|_2 = 2/N_j$ (the opposite level-$l_j$ child inside the same level-$(l_j - 1)$ ball), each of variance $N_j a_j / n$, contributing total row variance $a_j$. At widths $n < N_j$, no pair lies at distance $2/N_j$, so the shell is absent.

\begin{lemma}[Finite-extension lemma]\label{lem:extension}
Fix finitely many earlier active shells at dyadic scales $N_1 < \cdots < N_m$ with $\sum_{j=1}^m \sqrt{a_j} \le 1/64$. For a candidate dyadic width $n = 2^l > N_m$, add a new shell at distance $2/n$ with mass $a := 1/L$, where $L := \log n - \log\log n$, and write $M = B + S$ for the earlier shells and the added shell. Then as $n = 2^l \to \infty$ (with the earlier shells fixed),
\[
\Pr\bigl(\det(I - M) < 0\bigr) \longrightarrow \frac{1}{2}.
\]
\end{lemma}

\begin{proof}
The matrices $B$ and $S$ are independent, and $S$ consists of $n/2$ disjoint $2 \times 2$ blocks $\begin{psmallmatrix} 0 & X_r \\ Y_r & 0 \end{psmallmatrix}$ with all $n$ directed entries independent $\mathcal{N}(0, 1/L)$.

\emph{1. Earlier shells.}
Up to separate row and column permutations, shell~$j$ consists of $N_j$ disjoint square blocks of side $d_j = n/N_j$ with i.i.d.\ $\mathcal{N}(0, a_j/d_j)$ entries. Packing disjoint balls of radius $1/8$ in the ball of radius $9/8$ yields a $1/4$-net $\mathcal{N}$ of the unit sphere in $\mathbb{R}^{d_j}$ with $|\mathcal{N}| \le 9^{d_j}$. Approximating both unit vectors in a bilinear form by net vectors loses at most half the operator norm, so a Gaussian tail bound and union bound over $\mathcal{N} \times \mathcal{N}$ give
\[
\Pr\bigl(\|\text{block}\| > 8\sqrt{a_j}\bigr) \le 2 \cdot 9^{2 d_j} e^{-8 d_j}.
\]
Since $81 < e^8$ and the number of old blocks is fixed as $d_j \to \infty$, summing across shells gives $\|B\| \le 8\sum_j \sqrt{a_j} \le 1/8 \le 1/4$ with probability $1 - o(1)$. Moreover, every entry of $B$ has variance at most $K/n$ for the fixed constant $K := \max(\{1\} \cup \{N_j a_j\}_j)$.

\emph{2. Top-shell determinant sign and spectral gap.}
Let $q := \Pr(XY > 1)$ for independent $X, Y \sim \mathcal{N}(0, 1/L)$. In polar coordinates for $(\sqrt{L} X, \sqrt{L} Y)$, the angle is uniform and the squared radius is $\Exp(1/2)$, so for every $t > 0$,
\begin{equation}\label{eq:polar}
Q(t) := \Pr(XY > t) = \frac{1}{2\pi} \int_{\cos 2\theta > 0} e^{-Lt / \cos 2\theta}\,d\theta.
\end{equation}
Thus $q \le e^{-L} \to 0$. On $|\theta| \le 1/(4\sqrt{L})$, $\cos 2\theta \ge 1 - 2\theta^2 \ge 1 - 1/(8L)$ implies $1/\cos 2\theta \le 1 + 1/L$ for $L \ge 1$, yielding $q \ge c e^{-L}/\sqrt{L}$ for an absolute constant $c > 0$. Because $e^{-L} = (\log n)/n$, we have $nq \ge c\sqrt{\log n} \to \infty$. Since $\det(I - S) = \prod_{r=1}^{n/2}(1 - X_r Y_r)$ with independent factors,
\begin{equation}\label{eq:sign-S}
\Pr\bigl(\det(I - S) < 0\bigr) = \frac{1 - (1 - 2q)^{n/2}}{2} \longrightarrow \frac{1}{2}.
\end{equation}
Set $\delta := (\log n)^{-4}$. Because $z e^{-Ltz}$ decreases on $z \ge 1$ for $Lt \ge 1$, differentiating~\eqref{eq:polar} under the integral gives $|Q'(t)| \le L e^{-Lt}$ for $t \ge 1 - \delta$ and large~$n$. Hence $\Pr(\exists r : |1 - X_r Y_r| \le \delta) \le n \delta L e^{-L(1 - \delta)} = o(1)$. Also $\Pr(\max |S_{ij}| > 3) \le 2n e^{-9L/2} = o(1)$. On the intersection of these events, the $2 \times 2$ inverse formula yields $\|(I - S_r)^{-1}\| \le (1 + \max(|X_r|, |Y_r|))/\delta \le 4/\delta$ for every pair block~$r$.

\emph{3. Schur elimination on ordinary pairs.}
Call pair~$r$ exceptional if $\max(|X_r|, |Y_r|) > 1/2$; let $E$ collect both coordinates of every exceptional pair, $G$ be its complement, and $d := |E|$. Then $\mathbb{E}[d] \le 4n e^{-L/8}$, so $\Pr(d > n^{15/16}) = o(1)$ by Markov's inequality. The partition $(E, G)$ depends only on~$S$. On $\|B\| \le 1/4$, since $\|S_{GG}\| \le 1/2$, the bulk block
\[
V := I - S_{GG} - B_{GG}
\]
satisfies $\|V^{-1}\| \le 4$ and $\det V > 0$ (by continuously scaling $S_{GG} + B_{GG}$ from~$0$). Its Schur complement in $I - M$ is
\[
U = I - S_{EE} - R, \qquad R := B_{EE} + B_{EG} V^{-1} B_{GE}.
\]
Set $\eta := n^{-1/64}$ (so $n\eta^2 = n^{31/32}$). Condition on any $S$ with $d \le n^{15/16}$ (if $E = \emptyset$, the sign equality is immediate). For any unit vectors $x, y \in \mathbb{R}^E$, $x^\top B_{EE} y$ is centered Gaussian of variance at most $K/n$; the two-net bound gives
\[
\Pr(\|B_{EE}\| > \eta \mid S) \le 2 \cdot 9^{2d} \exp\bigl(-n\eta^2 / (8K)\bigr) = o(1)
\]
uniformly in~$S$. Next, condition further on $(B_{GG}, B_{GE})$ on the event that $V$ is invertible and $\|V^{-1} B_{GE}\| \le 1$ (implied by $\|B\| \le 1/4$, whose failure has probability $o(1)$). Setting $H := V^{-1} B_{GE}$ ($\|H\| \le 1$), independence of the directed entries of $B$ makes $B_{EG}$ independent of $(S, B_{GG}, B_{GE})$. Thus for fixed unit $x, y \in \mathbb{R}^E$, $x^\top B_{EG} H y$ is centered Gaussian of variance at most $(K/n)\|x\|_2^2 \|Hy\|_2^2 \le K/n$, and the same two-net bound gives $\Pr(\|B_{EG} H\| > \eta) = o(1)$. Hence $\|R\| \le 2n^{-1/64}$ with probability $1 - o(1)$.

Consequently, $\|(I - S_{EE})^{-1} R\| \le 8n^{-1/64}/\delta < 1$ for large~$n$ with probability $1 - o(1)$, so scaling $R$ continuously to~$0$ shows $\sgn\det U = \sgn\det(I - S_{EE})$. Because $\det V > 0$ and every ordinary pair has $1 - X_r Y_r \ge 3/4 > 0$, Schur's determinant identity $\det(I - M) = \det(V)\det(U)$ yields
\[
\Pr\bigl(\sgn\det(I - M) = \sgn\det(I - S)\bigr) \longrightarrow 1.
\]
Combining with~\eqref{eq:sign-S} completes the proof.
\end{proof}

\begin{theorem}[Fixed-profile refutation of \texttt{Conj\_random}]\label{thm:random}
For $p = 2$, there exist a deterministic, finite-valued, nonnegative profile $v$ with $v(|\cdot|_2) \in L^1(\mathbb{Z}_2)$ and $0 < \rho(v)^2 \le 1/4096$ ($\rho(v) \le 1/64 < 1$) and an increasing sequence of dyadic widths $N_k = 2^{l_k}$ such that
\[
\Pr_{N_k}\bigl(h = M^{(l_k)}\tanh h \text{ has at least } 3 \text{ equilibria}\bigr) \ge \frac{1}{2} - 2^{-k} \qquad (\forall k \ge 1).
\]
\end{theorem}

\begin{proof}
Set $N_0 := 1$. Recursively for $k \ge 1$, given $N_1 < \cdots < N_{k-1}$ with $\sum_{j=1}^{k-1} \sqrt{a_j} < 1/64$, apply Lemma~\ref{lem:extension} to select the least dyadic width $N_k = 2^{l_k} > N_{k-1}$ such that
\[
\sqrt{a_k} = (\log N_k - \log\log N_k)^{-1/2} \le \frac{2^{-k}}{64}, \qquad
\Bigl|\Pr_{N_k}\bigl(\det(I - M) < 0\bigr) - \frac{1}{2}\Bigr| \le 2^{-k}.
\]
Define $v(2/N_k) := N_k a_k$ for all $k \ge 1$ and $v(r) := 0$ elsewhere. Because later shells ($N_j > N_k$) are invisible at width $N_k$, every probability estimate persists for this single fixed profile~$v$. At every dyadic width~$n$, the normalized row-sum variance equals $\sum_{k : N_k \le n} a_k$, so the limit exists and satisfies
\[
0 < \rho(v)^2 = \sum_{k \ge 1} a_k = \int_{\mathbb{Z}_2} v(|x|_2)\,dx \le \Bigl(\sum_{k \ge 1} \sqrt{a_k}\Bigr)^2 \le \frac{1}{4096},
\]
since the shell $\{|x|_2 = 2/N_k\}$ has Haar measure $1/N_k$. Lemma~\ref{lem:degree} at each width $N_k$ gives at least three equilibria with probability at least $\frac{1}{2} - 2^{-k}$.
\end{proof}

\begin{remark}[Scope, bounded profiles, and relation to Wainrib--Touboul]
The profile $v$ in Theorem~\ref{thm:random} satisfies the literal hypotheses of Conjecture~\texttt{Conj\_random} ($v \ge 0$, finite everywhere, and $v(|\cdot|_2) \in L^1(\mathbb{Z}_2)$), and the budget $1/64$ can be made arbitrarily small. However, $v(2/N_k) = N_k a_k \to \infty$ as $k \to \infty$, so $v$ is unbounded near~$0$. Both the restriction of Conjecture~\texttt{Conj\_random} to bounded fixed profiles ($v \in L^\infty$) and the independent supercritical ($\rho(v) > 1$) exponential-growth implication remain open. Note also that the constant-profile result of Wainrib and Touboul~\cite{wainrib2013} cited in~\cite{zuniga2026} (cf.~\cite{sompolinsky1988}) establishes exponential growth of the \emph{expected} equilibrium count rather than a high-probability lower bound.
\end{remark}
