\begin{abstract}
We refute the necessity direction of Conjecture~4.8 (\texttt{conj:dichotomy}) in Qi (\href{https://arxiv.org/abs/2609.27384}{arXiv:2609.27384}, September 2026) and establish an exact replacement characterization.
First, under clause~\textup{(C-a)} of Anchor~C alone, both the tail shape index $\alpha$ and the tail scale $L$ are automatically identifiable on every candidate fiber $\mathcal{M}_{\mathcal{K}}(\ell_0)$.
Second, we construct an explicit one-parameter scaled-uniform candidate family $\{K_\theta : \theta \in [1, 2]\}$ with strictly positive bounded widths and no escape to infinity that satisfies~\textup{(C-a)} and identifies $(\alpha, L, p_\tau)$ on every fiber $\mathcal{M}_{\mathcal{K}}(\ell_\theta)$, yet fails the fixed-scale injectivity clause~\textup{(C-b)} on every fiber---even on regular pairs inside $\mathcal{M}_{\mathcal{K}}(\ell_\theta)$.
Third, we prove that under~\textup{(C-a)}, fiberwise identifiability of $(L, p_\tau)$ holds if and only if for every $\theta$ with nonempty fiber, every finite signed measure $\nu \in \ker K_\theta$ supported on $[0, \bar s_{\mathcal{K}}]$ with $\nu(\{0\}) = 0$ and vanishing density on $(0, \bar s_0)$ satisfies $\nu([0, \tau]) = 0$.
\end{abstract}

\section{Setup and Forward Implication}

We follow Definitions~\texttt{def:M}, \texttt{def:psi}, Assumption~\texttt{ass:C} (Anchor~C), and \texttt{def:fiber} of Qi~\cite{Qi2026}, retaining $\bar s_0$ for the shared near-zero density floor and $\bar s_{\mathcal{K}}$ for the candidate family's finite severity cap (so as not to collide with the Euler Beta function $B(\cdot, \cdot)$ in~\cite{Qi2026}).

A model member is a pair $(F_S, K)$ consisting of a Borel probability law $F_S$ on $[0, \infty)$ and a Borel Markov kernel $K(\cdot \mid s)$, observed only through the marginal mixture $Q = K F_S := \int K(\cdot \mid s)\,F_S(ds)$.
Write $\ell(s)$ and $u(s)$ for the finite lower and upper support endpoints of $K(\cdot \mid s)$, and $w(s) := u(s) - \ell(s)$ for its width.
In the regularity class $\Mreg$ with shared class constants $(\bar s_0, s_A, s_R, \tau, \beta, c_-, c_+, \bar b)$ obeying $0 < \tau < s_R \le \bar s_0$:
\begin{itemize}[leftmargin=1.6em]
\item \textbf{A1:} $F_S$ has lower endpoint~$0$, $F_S(\{0\}) = 0$, and a density version on $(0, \bar s_0]$ satisfying $f_S(s) = L s^\alpha(1 + o(1))$ as $s \downarrow 0$ with member-specific $L > 0$ and $\alpha \ge 0$. The target estimand is $\psi = (\alpha, L, p_\tau)$ where $p_\tau := F_S([0, \tau])$.
\item \textbf{A2--A4, NA:} $0 < w(s) < \infty$ everywhere with no escape to infinity on $[0, s_A]$ (A2); the midpoint bias $m(s) := (\ell(s) + u(s))/2 - s$ satisfies $|m(s)| \le \bar b$ and $m \not\equiv 0$ (A3); $w(s)/s \to \infty$ as $s \downarrow 0$ (A4); and $K(\cdot \mid s)$ is not $F_S$-almost everywhere an additive error law (NA).
\item \textbf{A5--A6:} $\ell$ is nondecreasing on $[0, \infty)$ with secant slopes in $[c_-, c_+]$ on $[0, s_R]$ (A5); and $K([\ell(s), \ell(s) + h] \mid s) = c_K(s) h^{\beta+1}(1 + o(1))$ uniformly on $s \in (0, s_R]$ as $h \downarrow 0$, with $c_K(s)$ bounded above and away from zero near~$0$ (A6).
\end{itemize}

A \emph{candidate family} $\mathcal{K} = \{K_\theta : \theta \in \Theta\}$ ($\Theta \subseteq \mathbb{R}^k$) has strictly increasing lower edges $\ell_{K_\theta}$ on $[0, \infty)$, a finite severity cap $\bar s_{\mathcal{K}} < \infty$, and satisfies clause \textbf{(C-a)}: all non-$\theta$ structure and the edge map $\ell_0 = \ell_{K_\theta}$ are known, and $\ell_{K_\theta}$ uniquely determines $\theta$.
Each admissible edge $\ell_0$ defines the fiber
\[
\mathcal{M}_{\mathcal{K}}(\ell_0) := \bigl\{(F_S, K) \in \Mreg : K \in \mathcal{K},\; \ell_K = \ell_0,\; \supp F_S \subseteq [0, \bar s_{\mathcal{K}}]\bigr\}.
\]
Clause \textbf{(C-b)} (\emph{fixed-scale injectivity}) requires that for every $\theta \in \Theta$, the mixture map $F \mapsto K_\theta F$ be injective on \emph{all} Borel probability measures supported on $[0, \bar s_{\mathcal{K}}]$.
Conjecture~4.8 (\texttt{conj:dichotomy}) of~\cite{Qi2026} asserts that for every candidate family $\mathcal{K}$, $(L, p_\tau)$ is identifiable on every fiber $\mathcal{M}_{\mathcal{K}}(\ell_0)$ if and only if $\mathcal{K}$ satisfies~\textup{(C-b)}.

As noted in~\cite{Qi2026}, sufficiency is immediate: under~\textup{(C-a)}, two members of the same fiber share the same kernel $K_\theta$; under~\textup{(C-b)}, $K_\theta F = K_\theta F'$ implies $F = F'$, which determines both $p_\tau$ and $(\alpha, L)$ (via $F([0, t]) \sim L t^{\alpha+1}/(\alpha+1)$ as $t \downarrow 0$).

\section{Automatic Identification of \texorpdfstring{$(\alpha, L)$}{(alpha, L)} Under (C-a)}

\begin{proposition}\label{prop:tail-ident}
Let $\mathcal{K}$ be any candidate family (satisfying \textup{(C-a)} only), and let $\mathcal{M}_{\mathcal{K}}(\ell_0)$ be a nonempty fiber with common kernel $K$. If $(F, K), (F', K) \in \mathcal{M}_{\mathcal{K}}(\ell_0)$ satisfy $KF = KF'$, then their \textup{A1} parameters coincide: $(\alpha, L) = (\alpha', L')$.
\end{proposition}

\begin{proof}
Suppose $(\alpha, L) \neq (\alpha', L')$.
If $\alpha \neq \alpha'$, then $f(s)/f'(s)$ tends to $0$ or $\infty$ as $s \downarrow 0$; if $\alpha = \alpha'$ and $L \neq L'$, then $f(s)/f'(s) \to L/L' \neq 1$.
Interchanging $F$ and $F'$ if necessary, there exists $d \in (0, s_R)$ such that $f(s) > f'(s)$ for all $s \in (0, d)$.

Choose $t \in (0, d)$ and consider $E := [0, \ell(t)]$.
Because $\ell = \ell_0$ is strictly increasing on $[0, \infty)$ by the standing structure of candidate families in Assumption~\texttt{ass:C}:
\begin{itemize}[leftmargin=1.6em]
\item For every $s > t$, $\ell(s) > \ell(t)$, so $K(E \mid s) = 0$.
\item For every $s \in (0, t)$, $\ell(s) < \ell(t)$, so $E$ contains a neighborhood of the finite lower support endpoint $\ell(s)$ and therefore $K(E \mid s) > 0$.
\end{itemize}
By~A1, $F(\{0\}) = F'(\{0\}) = 0$, and both laws have densities on $(0, s_R]$ so $\{t\}$ is null under both. Consequently,
\[
(KF)(E) - (KF')(E) = \int_0^t K(E \mid s)\bigl(f(s) - f'(s)\bigr)\,ds > 0,
\]
contradicting $KF = KF'$.
\end{proof}

\section{Scaled-Uniform Counterexample to the Necessity of (C-b)}

\begin{theorem}[Counterexample to Conjecture~4.8]\label{thm:counterexample}
Fix shared class constants $\bar s_{\mathcal{K}} = 3$, $s_A = s_R = \bar s_0 = 1$, $\tau = 1/2$, $\beta = 0$, $c_- = c_+ = 1$, and $\bar b = 4$. Define the Borel width functions
\[
w_0(s) := 4 - \min(s, 3), \qquad
w(s) := \begin{cases}
s - 2, & 2 < s \le 3,\\
w_0(s), & \text{otherwise},
\end{cases}
\]
and for $\theta \in \Theta := [1, 2]$ define $K_\theta(\cdot \mid s) := \Law(\theta + s + w(s) U)$ with $U \sim \Unif[0, 1]$. Then:
\begin{enumerate}[label=\textup{(\arabic*)}]
\item $\mathcal{K} = \{K_\theta : \theta \in [1, 2]\}$ is a candidate family satisfying \textup{(C-a)}, with every fiber $\mathcal{M}_{\mathcal{K}}(\ell_\theta)$ nonempty and realizing non-constant values of $(L, p_\tau)$.
\item The full estimand $(\alpha, L, p_\tau)$ is identifiable on every fiber $\mathcal{M}_{\mathcal{K}}(\ell_\theta)$.
\item Clause \textup{(C-b)} fails on every fiber $\mathcal{M}_{\mathcal{K}}(\ell_\theta)$, even on regular pairs $(F_1, K_\theta)$ and $(F_2, K_\theta)$ inside $\mathcal{M}_{\mathcal{K}}(\ell_\theta)$.
\end{enumerate}
\end{theorem}

\begin{proof}
\emph{Step 1: Admissibility and \textup{(C-a)}.}
The map $(s, u) \mapsto \theta + s + w(s)u$ is jointly Borel measurable, so $K_\theta$ is a Borel Markov kernel. Its lower edge $\ell_\theta(s) = \theta + s$ is strictly increasing on $[0, \infty)$ and uniquely determines $\theta = \ell_\theta(0)$, establishing~\textup{(C-a)}. For every $\theta \in [1, 2]$:
\begin{itemize}[leftmargin=1.6em]
\item \emph{A2:} $0 < w(s) \le 4$ for all $s \ge 0$ (specifically $w(s) \in [1, 4]$ outside $(2, 3]$ and $w(s) = s - 2 \in (0, 1]$ on $(2, 3]$), with no escape on $[0, 1]$.
\item \emph{A3:} The midpoint bias is $m_\theta(s) = \theta + w(s)/2 \in (1, 4]$, so $0 < |m_\theta(s)| \le 4 = \bar b$ everywhere.
\item \emph{A4:} On $(0, 1]$, $w(s) = 4 - s \to 4$, so $w(s)/s \to \infty$ as $s \downarrow 0$.
\item \emph{A5:} $\ell_\theta(s) = \theta + s$ has constant secant slope $1 \in [c_-, c_+]$.
\item \emph{A6:} For every $s \in (0, 1]$ and $h \in (0, 3]$, the edge mass is identically $K_\theta([\ell_\theta(s), \ell_\theta(s) + h] \mid s) = h/(4 - s)$, which verifies uniform~A6 with $\beta = 0$ and coefficient $c_K(s) = 1/(4 - s) \in [1/4, 1/3]$.
\item \emph{NA:} Every A1 law $F$ puts positive density on an interval $(0, \delta) \subseteq (0, 1)$ where $w(s) = 4 - s$ is strictly decreasing, whereas any fixed additive error law has constant width; hence NA holds for every A1 law paired with $K_\theta$.
\end{itemize}

\emph{Step 2: Fiberwise recovery of $(\alpha, L, p_\tau)$.}
Fix $\theta \in [1, 2]$ and $(F, K_\theta) \in \mathcal{M}_{\mathcal{K}}(\ell_\theta)$.
For $t \in (0, 1)$, only severities $s \le t$ satisfy $\ell_\theta(s) \le \theta + t$, and for every $s \in [0, t] \subset [0, 1)$ the upper endpoint is $\theta + s + (4 - s) = \theta + 4 > \theta + t$.
By Tonelli's theorem, $Q = K_\theta F$ has a Lebesgue density on $(\theta, \theta + 1)$ with right-continuous version
\[
H(t) := \int_{[0, t]} \frac{F(ds)}{4 - s} \qquad (0 < t < 1).
\]
Right continuity determines $H$ uniquely on $(0, 1)$ from $Q$. Its Lebesgue--Stieltjes measure on $(0, 1)$ satisfies $(4 - s)\,dH(s) = F(ds)$, so $Q$ uniquely recovers $F|_{(0, 1)}$. Combined with $F(\{0\}) = 0$ from~A1, $Q$ uniquely determines $p_\tau = F([0, 1/2])$ and $(\alpha, L)$.

\emph{Step 3: Failure of \textup{(C-b)} on every fiber.}
Set $s_n := 2 + 2^{1 - n} \in (2, 3]$ for $n \ge 1$ and $G := \sum_{n=1}^\infty 2^{-n}\delta_{s_n}$, which is supported on $[0, 3]$.
At $s = 2$, $K_\theta(\cdot \mid 2) = \Unif[\theta + 2,\, \theta + 4]$ has density $1/2$ on $(\theta + 2, \theta + 4)$.
At $s_n$, $w(s_n) = 2^{1 - n}$, so $K_\theta(\cdot \mid s_n) = \Unif[\theta + 2 + 2^{1 - n},\, \theta + 2 + 2^{2 - n}]$.
These intervals tile $\theta + (2, 4]$ up to endpoints, and weighting the $n$-th tile of length $2^{1 - n}$ by $2^{-n}$ yields uniform density $1/2$ on every tile. Hence
\[
K_\theta G = K_\theta \delta_2 = \Unif[\theta + 2,\, \theta + 4],
\]
while $G(\{2\}) = 0 \neq 1 = \delta_2(\{2\})$.

To realize non-injectivity inside the regular fiber $\mathcal{M}_{\mathcal{K}}(\ell_\theta)$, let $D := \Unif[0, 1]$ and for any $r \in (0, 1)$ set
\[
F_1 := r D + (1 - r)\delta_2, \qquad F_2 := r D + (1 - r)G.
\]
Both laws are supported on $[0, 3]$ and have constant density $r$ on $(0, 1)$, so both satisfy~A1 with $(\alpha, L, p_\tau) = (0, r, r/2)$ and therefore belong to $\mathcal{M}_{\mathcal{K}}(\ell_\theta)$. Yet $K_\theta F_1 = K_\theta F_2$ while $F_1 \neq F_2$. Letting $r$ range over $(0, 1)$ also shows that $(L, p_\tau)$ varies nontrivially on every fiber.
\end{proof}

\begin{remark}[Jump discontinuity and injective comparison family]
The width map $w$ in Theorem~\ref{thm:counterexample} has a single jump discontinuity at $s = 2$ (strictly outside $[0, s_R] = [0, 1]$). This is fully permitted because $\Mreg$ and Assumption~\texttt{ass:C} require only Borel measurability of the Markov kernel and bounded positive widths, never continuity of $w$.
By comparison, using $w_0(s) = 4 - \min(s, 3)$ everywhere defines a companion candidate family $\{K_\theta^0 : \theta \in [1, 2]\}$ satisfying the same regularity axioms in which every conditional support on $[0, 3]$ ends at $\theta + 4$. Then $K_\theta^0 F$ has right-continuous density $H(t) = \int_{[0, \min(t, 3)]} (4 - s)^{-1}\,F(ds)$ on $[0, 4)$ (and $0$ for $t < 0$), so $(4 - s)\,dH(s) = F(ds)$ on $[0, 3]$---including possible atoms at $0$ and $3$ via the jumps of $H$---and $\{K_\theta^0\}$ satisfies~\textup{(C-b)}.
\end{remark}

\section{Exact Necessary and Sufficient Replacement for (C-b)}

Given a candidate family $\mathcal{K} = \{K_\theta : \theta \in \Theta\}$ with severity cap $\bar s_{\mathcal{K}}$ and shared density floor $\bar s_0$, let $\mathcal{V}$ be the linear space of finite signed Borel measures $\nu$ supported on $[0, \bar s_{\mathcal{K}}]$ satisfying
\[
\nu(\{0\}) = 0, \qquad
\nu|_{(0, \bar s_0)} = g(s)\,ds, \qquad
\lim_{s \downarrow 0} g(s) = 0
\]
for some density version $g$. Extend $K_\theta$ linearly to signed measures via $(K_\theta \nu)(A) := \int K_\theta(A \mid s)\,\nu(ds)$.

\begin{theorem}[Exact characterization of fiberwise scale identifiability]\label{thm:char}
Let $\mathcal{K} = \{K_\theta : \theta \in \Theta\}$ be a candidate family (satisfying \textup{(C-a)}). Then $(L, p_\tau)$ is identifiable on every fiber $\mathcal{M}_{\mathcal{K}}(\ell_0)$ if and only if for every $\theta \in \Theta$ whose fiber $\mathcal{M}_{\mathcal{K}}(\ell_{K_\theta})$ is nonempty,
\begin{equation}\label{eq:exact-cond}
\nu \in \mathcal{V} \quad \text{and} \quad K_\theta \nu = 0 \quad \Longrightarrow \quad \nu([0, \tau]) = 0.
\end{equation}
\end{theorem}

\begin{proof}
\emph{Sufficiency.} Assume~\eqref{eq:exact-cond}, and let $(F, K_\theta), (F', K_\theta) \in \mathcal{M}_{\mathcal{K}}(\ell_{K_\theta})$ satisfy $K_\theta F = K_\theta F'$.
By Proposition~\ref{prop:tail-ident}, $F$ and $F'$ share the same A1 parameters $(\alpha, L)$ with $\alpha \ge 0$.
Put $\nu := F - F'$. Then $\supp \nu \subseteq [0, \bar s_{\mathcal{K}}]$, $\nu(\{0\}) = 0$, $K_\theta \nu = 0$, and on $(0, \bar s_0)$ its density $f(s) - f'(s) = o(s^\alpha)$ tends to~$0$ as $s \downarrow 0$ because $\alpha \ge 0$.
Thus $\nu \in \mathcal{V} \cap \ker K_\theta$, and~\eqref{eq:exact-cond} gives $F([0, \tau]) - F'([0, \tau]) = \nu([0, \tau]) = 0$.

\emph{Necessity.} Suppose there exist $\theta \in \Theta$ with nonempty fiber $\mathcal{M}_{\mathcal{K}}(\ell_{K_\theta})$ and $\nu \in \mathcal{V}$ such that $K_\theta \nu = 0$ and $\nu([0, \tau]) \neq 0$.
Because $K_\theta(\mathbb{R} \mid s) = 1$ for all $s$, evaluating $(K_\theta \nu)(\mathbb{R}) = 0$ forces $\nu([0, \bar s_{\mathcal{K}}]) = 0$.
In the Jordan decomposition $\nu = \nu^+ - \nu^-$, both parts are supported on $[0, \bar s_{\mathcal{K}}]$ with equal positive total mass $m := \nu^+([0, \bar s_{\mathcal{K}}]) = \nu^-([0, \bar s_{\mathcal{K}}]) > 0$, no atom at~$0$, and densities $g^\pm(s) = \max(\pm g(s), 0)$ on $(0, \bar s_0)$ satisfying $\lim_{s \downarrow 0} g^\pm(s) = 0$.

Pick one member $(F_*, K_\theta) \in \mathcal{M}_{\mathcal{K}}(\ell_{K_\theta})$. Nonemptiness forces $\bar s_{\mathcal{K}} > 0$ (since $F_*$ has lower endpoint~$0$ and no atom at~$0$). Choose $d \in (0, \min(\bar s_{\mathcal{K}}, \bar s_0))$, let $D := \Unif[0, d]$, and define the probability measures
\[
F_+ := \frac{1}{3}\bigl(F_* + D + \nu^+/m\bigr), \qquad
F_- := \frac{1}{3}\bigl(F_* + D + \nu^-/m\bigr).
\]
Both are supported on $[0, \bar s_{\mathcal{K}}]$, have no atom at~$0$, and possess densities on $(0, \bar s_0]$.
As $s \downarrow 0$, the density of $F_*$ tends to $L_* \mathbf{1}_{\{\alpha_* = 0\}} \in [0, \infty)$, the uniform component $D$ contributes $1/d > 0$ on $(0, d)$, and $g^\pm(s)/m \to 0$.
Hence both $F_+$ and $F_-$ satisfy~A1 with exponent $\alpha = 0$ and positive scale $(L_* \mathbf{1}_{\{\alpha_* = 0\}} + 1/d)/3$.
The kernel-only axioms A2--A6 are inherited from $(F_*, K_\theta)$.
Moreover, $F_* \ll F_\pm$, so any additive error representation valid $F_\pm$-a.e.\ would also hold $F_*$-a.e., contradicting~NA for $(F_*, K_\theta)$.
Thus $(F_+, K_\theta), (F_-, K_\theta) \in \mathcal{M}_{\mathcal{K}}(\ell_{K_\theta})$, yet $F_+ - F_- = \nu/(3m)$ implies $K_\theta F_+ = K_\theta F_-$ and $F_+([0, \tau]) - F_-([0, \tau]) = \nu([0, \tau])/(3m) \neq 0$.
\end{proof}

\section{Exact Computational Verification}

The script \texttt{solutions/nonadd\_kernel\_ident/explore.py} checks the scaled-uniform families of Theorem~\ref{thm:counterexample} and Remark~1 in exact rational arithmetic (\texttt{fractions.Fraction}) across $\theta \in \{1, 3/2, 2\}$.
It performs $1{,}578$ exact mixture CDF comparisons confirming $K_\theta G = K_\theta \delta_2$ (summing the full infinite dyadic tail analytically), $20$ exact local recovery checks for polynomial-weighted latent densities on $(0, 1)$, and capped atomic inversions at $s \in \{0, 1/2, 2, 3\}$ for the injective family $w_0$. All assertions pass.
